```go
  p.X = x

  fmt.Println(p)
  p.SetPoint(2,3)

```

### 指针

```go
type Point struct{
  X int
  Y int
}
func main (){
  p := Point{1,2}
  fmt.Println(p)
  q := p
  q.X = 5
  // 此时赋值，两者不会干扰（默认按值赋值）
  fmt.Println(q) // 5,2
  fmt.Println(p) // 1,2
  
  w := &p
  w.X = 10
  fmt.Println(*w) // 10,2
  fmt.Println(p) // 10,2
}
```

### 接口

```go
type Shape interface {
  Print()
}
type Rectangle struct {}
type Triangle struct {}
func (r Rectangle) Print(){
  fmt.Println("矩形")
}
func (t Triangle) Print(){
  fmt.Println('三角形')
}

func  main(){
  var s Shape
  s = Rectangle{}
  s.Print() // 矩形
  s = Triangle{}
  s.Print() // 三角形
}
```

### 并发

```go

  time.Sleep(500* time.Milliscond)
  fmt.Println("func1")
}
func func2(){
  fmt.Println("func2")
}

func main(){
  // go 关键字用来创建一个新的线程来执行内容
  go func2()
  func1()
}
```

Channels

```go
func func1(ch chan string){
  ch <- "函数1"
}
func func2(ch chan string){
  ch <- "函数1"
}
func main(){
  ch := make(chan string)
  go func2(ch)
  res1 := <-ch
  go func1(ch)
  res2 := <-ch
  fmt.Println(res1,res2 )
}
```
